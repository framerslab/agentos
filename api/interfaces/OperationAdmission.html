Interface OperationAdmission

An admission bound to an operation: the whole amount for every run, run 1's bound, and run 1's lease. The runs' lease fences the row, so an operation admission never takes requireFence.

interface OperationAdmission {
    kind: string;
    micro: number;
    at: Date;
    expires: Date;
    capMicro: number;
    requireFence?: boolean;
    operation: string;
    runMicro: number;
    leaseUntil: Date;
}
