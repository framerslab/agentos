Interface WikiAttachableMemory

Minimal Memory-facade surface this helper needs (structural: the facade's types alone are imported, never the class): the trace calls the operator's pages are indexed with, and the page rows a subject's own pages are kept in (UserWikiPagesPort).

interface WikiAttachableMemory {
    pageExistsOutside(scopeId, pageId): Promise<boolean>;
    remember(content, options?): Promise<{
        id: string;
    }>;
    forget(traceId): Promise<void>;
    deleteTraces(ids): Promise<unknown>;
    attachWiki(wiki): void;
    readPageRow(scopeId, pageId): Promise<null | {
        content: string;
        contentHash: string;
        chunkIds: string[];
    }>;
    listPageRows(scopeId): Promise<{
        pageId: string;
        content: string;
        contentHash: string;
        chunkIds: string[];
    }[]>;
    writePageRow(scopeId, page): Promise<{
        chunkIds: string[];
    }>;
    reindexPageRow(scopeId, pageId, chunksOf): Promise<null | {
        chunkIds: string[];
        removed: string[];
    }>;
    deletePageRows(scopeId, pageId?): Promise<{
        pages: number;
        chunks: number;
    }>;
}
